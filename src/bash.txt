function updateMainMap() {

      return;
    }

    const popup = `<div style="font-family:Vazirmatn,sans-serif;direction:rtl;min-width:140px">

function generateGPS(base) {
  return {
    lat: base.lat + (Math.random() - 0.5) * 0.002,
    lng: base.lng + (Math.random() - 0.5) * 0.002,
    speed: Math.floor(Math.random() * 120),
    satellites: Math.floor(Math.random() * 8) + 4,
    signal: Math.random() > 0.08 ? 'ok' : 'lost',
    engine: Math.random() > 0.25,
    timestamp: Date.now(),
  };
}

function startSimulator() {
  // initial data
  vehicles = vehicles.map((v, i) => {
    const gps = generateGPS(BASE_POS[i]);
    return { ...v, gps, history: [{ ...gps, connected: true, time: now() }] };
  });
  updateDashboard();
  updateMainMap();

  simInterval = setInterval(() => {
    vehicles = vehicles.map((v, i) => {
      const gps = generateGPS(BASE_POS[i]);
      const point = { ...gps, connected: Math.random() > 0.15, time: now() };
      return { ...v, gps, history: [...v.history.slice(-99), point] };
    });
    updateDashboard();
    updateMainMap();
    if (currentPage === 'history') updateHistoryPage();
  }, 3000);
}
